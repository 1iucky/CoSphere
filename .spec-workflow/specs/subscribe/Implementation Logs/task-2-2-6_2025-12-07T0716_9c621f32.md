// CreateSubscriptionRedemption:450-512 - 添加参数校验
func CreateSubscriptionRedemption(
    name string,
    planId int64,
    duration int64,
    durationDays int,
    pricePaid int64,
    redeemOption string,
    couponId *int64,
    expiredTime int64,
    creatorId int,
) (*Redemption, error) {
    // ========== 参数校验 ==========
    // 1. 套餐 ID 不能为空
    if planId == 0 {
        return nil, errors.New("套餐 ID 不能为空")
    }

    // 2. 验证兑换选项合法性（必须是 stack/coexist/convert/replace/extend 之一）
    validOptions := map[string]bool{
        common.RedeemOptionStack:   true,
        common.RedeemOptionCoexist: true,
        common.RedeemOptionConvert: true,
        common.RedeemOptionReplace: true,
        common.RedeemOptionExtend:  true,
    }
    if !validOptions[redeemOption] {
        return nil, errors.New("无效的兑换选项：必须是 stack/coexist/convert/replace/extend 之一")
    }

    // 3. 验证时长：duration 或 durationDays 至少有一个 > 0
    if duration <= 0 && durationDays <= 0 {
        return nil, errors.New("时长无效：duration 或 durationDays 至少有一个必须 > 0")
    }

    // 4. 验证价格（不能为负数）
    if pricePaid < 0 {
        return nil, errors.New("价格不能为负数")
    }

    // ========== 构造 Payload ==========
    payload := RedemptionSubscriptionPayload{
        PlanId:       planId,
        Duration:     duration,
        DurationDays: durationDays,
        PricePaid:    pricePaid,
        RedeemOption: redeemOption,
        CouponId:     couponId,
    }

    // 序列化 payload
    payloadBytes, err := json.Marshal(payload)
    if err != nil {
        return nil, err
    }
    payloadStr := string(payloadBytes)

    // ========== 创建 Redemption ==========
    redemption := &Redemption{
        Name:        name,
        Type:        common.RedemptionTypeSubscription,
        Payload:     &payloadStr,
        Status:      common.RedemptionCodeStatusEnabled,
        ExpiredTime: expiredTime,
        UserId:      creatorId,
        Key:         GetRandomString(32),
        CreatedTime: time.Now().Unix(),
    }

    err = DB.Create(redemption).Error
    return redemption, err
}
